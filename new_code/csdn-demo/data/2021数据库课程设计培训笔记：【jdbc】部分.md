```
package myNote;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.sql.Statement;
public class MainDemo {

	public static void main(String[] args) throws ClassNotFoundException, SQLException
	{
		//1、加载驱动
		Class.forName("com.mysql.jdbc.Driver");
		//2、建立连接
		//url:数据库连接路径
		//user：用户名
		//password：密码
		String url = "jdbc:mysql://localhost:3306/practice";//practice是数据库的库名
		String user = "root";//用户名是自己数据库的
		String password = "1234";//密码是自己的
		Connection conn = DriverManager.getConnection(url,user,password); //与数据库建立联系
		//3、获得statement对象
		Statement s = conn.createStatement();//statement对象用于将sql语句发送到数据库中
		//4.用statement对象将sql语句传入数据库执行操作举例
		String sql = "select * from people";//people是表名
		ResultSet rs = s.executeQuery(sql);//ResultSet为结果集，存放查询到的结果
		System.out.println("查询成功");
	}
}
```

##### 2.2.1 基础操作举例

**注意：java中方法不能直接访问方法，必须通过对象访问，或者均声明为静态方法。** 可参考：[java主类的main方法调用其他方法](https://www.cnblogs.com/baby-zhude/p/7918834.html)  
 *个人感觉：因为这个规定比较麻烦，所以常见的写法总是把main方法单独写在一个类，通过对象访问外部类的方法。*  
 **工具类：**

```
package myNote;

import java.sql.ResultSet;
import java.sql.SQLException;

public class FindDemo {

	public void find(ResultSet rs) throws SQLException
	{
		int i = 0;
		while(rs.next())
		{
			//rs.next()是找结果集中下一个结果，返回值是bool类型，能找到为true，找不到为false
			//用rs.get方法分别获得每列数据，根据数据类型不同，设置不同类型的变量和get方法的写法
			int pid = rs.getInt(1);
			String pname = rs.getString(2);//获得第二列的数据
			String pEname = rs.getString(3);//获得每列数据的语句可以前后调换，不必按照顺序
			i++;
			System.out.println("查到的第"+i+"条数据");
			System.out.println("pid="+pid+",pname="+pname+",pEname="+pEname);
		}
	}
}
```

##### 2.2.2 封装工具类

因为建立数据库的连接和操作方法语句复杂且之间又存在一定的联系，通过工具类封装常用方法实现反复调用  
 **工具类：**

```
package myNote;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.sql.Statement;

public class Tools {

	//获取连接对象
	public Connection getConn() throws ClassNotFoundException, SQLException 
	{
		Class.forName("com.mysql.jdbc.Driver");
		String url = "jdbc:mysql://localhost:3306/practice";
		String user = "root";
		String password = "1234";
		return DriverManager.getConnection(url,user,password);
	}
	//创建statement对象
	public Statement getStat() throws ClassNotFoundException, SQLException
	{
		return getConn().createStatement();
	}
	//获得数据集(方法较为灵活)
	//获得people表中全部数据
	public ResultSet getAll() throws ClassNotFoundException, SQLException
	{
		String sql = "select * from people";
		return getStat().executeQuery(sql);
	}
	//获得people表中特定数据
	public ResultSet getOne() throws ClassNotFoundException, SQLException
	{
		String sql = "select * from people where pname = '张三' ";
		//注意数据库语言，字符串用单引号
		return getStat().executeQuery(sql);
	}
	public ResultSet getId(int id) throws ClassNotFoundException, SQLException
	{
		String sql = "select * from people where pid = " + id ;
		return getStat().executeQuery(sql);
	}
	//打印结果方法
	public void printresult(ResultSet result) throws SQLException
	{
		while(result.next())
		{
			int pid = result.getInt(1);
			String pname = result.getString(2);
			String pEname = result.getString(3);
		   	System.out.println("pid="+pid+",pname="+pname+",pEname="+pEname);
		}
	}
}
```

**注意：** `result.next()`是很重要的，如果没有正确使用，可能导致`result.get()`也失败。  
 提示：注意数据库语言的写法。举例：传指定姓名为参数的查询语句：

```
public ResultSet getName(String name) throws ClassNotFoundException, SQLException
		{
			String sql = "select * from people where pname = " + "'" + name + "'";
			return getStat().executeQuery(sql);
		}
```

##### 2.3.2 特殊方法

**工具类：**

```
package myNote;

import java.sql.*;

public class Tools {

	//获取连接对象
	public Connection getConn() throws ClassNotFoundException, SQLException 
	{
		Class.forName("com.mysql.jdbc.Driver");
		String url = "jdbc:mysql://localhost:3306/practice";
		String user = "root";
		String password = "1234";		
		return DriverManager.getConnection(url,user,password);
	}
	//设置简单的字符串拼接方法,创建另一种statement对象
	public PreparedStatement simplyWay(String sql) throws ClassNotFoundException, SQLException
	{
			
		Connection conn = getConn();
		return conn.prepareStatement(sql);
	}
	//用简单的字符串拼接方法插入一条信息
	public int InsertOne(int pid, String pname, String Pename) throws ClassNotFoundException, SQLException
	{
		String sql = "insert into people values(?,?,?)";
		PreparedStatement ps = simplyWay(sql);
		ps.setInt(1, pid);
		ps.setString(2, pname);
		ps.setString(3, Pename);
		return ps.executeUpdate();
	}
}
```
