我的思路是这样的
--------

如果我们有高为5的矩形，计算面积肯定按照5*1而不是3*1算，同理，不管怎么找矩形，一定是按照木板效应，就是有一个高度正好在限制的，不可能明明可以取到5，我们非要取3。换句话说，我最后选择的大矩形高度一定是曾经出现过的。  
那我的想法就是，把每一个候选高度最长的连续宽记录下来，然后再遍历这些候选高度，找到最大的高\*宽就是矩形。  
每添加一个矩形，就更新现在的数据。把目前能连着的高度放在队列里，（曾经能连但是现在已经连不起来的就删掉了，删掉的时候比较目前和曾经最大连续宽并更新）。具体我不说了，代码虽然有点复杂，但大概看一下能明白应该。满分代码如下：

```
#include<cstdio>
#include<queue>
#include<iostream>
using namespace std;
int ans[5][100005];
// 0. 现在队列中的all
// 1. 最多的max
int main()
{
    queue <int> nowq;
    int n;
    cin>>n;
    int maxx = 0;
    while(n--)
    {
        int high;
        cin>>high;
        if(nowq.empty())
        {
            nowq.push(high);
            ans[0][high] = 1;
        }
        else
        {
            int lenth = nowq.size();
            int flag = 0;
            int flag_max = 1;
            while(lenth--)
            {
                int high_bef = nowq.front();
                nowq.pop();
                if(high > high_bef)
                {
                    ans[0][high_bef]++;
                    nowq.push(high_bef);
                }
                else if(high == high_bef)
                {
                    flag = 1;
                    ans[0][high_bef]++;
                    nowq.push(high_bef);
                }
                else
                {
                    if(ans[0][high_bef] + 1 > flag_max)
                    {
                        flag_max = ans[0][high_bef] + 1;
                    }
                    if(ans[0][high_bef] > ans[1][high_bef])
                    {
                        ans[1][high_bef] = ans[0][high_bef];
                    }
                    ans[0][high_bef] = 0;
                }
            }
            if(flag == 0)
            {
                nowq.push(high);
                ans[0][high] = flag_max;
            }
        }
    }
    while(!nowq.empty())
    {
        int high_bef = nowq.front();
        nowq.pop();
        if(ans[1][high_bef] < ans[0][high_bef])
        {
            ans[1][high_bef] = ans[0][high_bef];
        }
    }
    for(int i = 1; i <= 100000; i++)
    {
        if(ans[1][i]*i > maxx)
        {
            maxx = ans[1][i]*i;
        }
        ans[0][i] = 0;
        ans[1][i] = 0;
    }
    cout<< maxx<< endl;
    return 0;
}
```

问题
--
